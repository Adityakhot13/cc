from flask import Flask, request

app = Flask(__name__)

@app.route('/')
def home():
    return '''
    <h2>Generate N Even Numbers</h2>
    <form action="/even" method="get">
        Enter N:
        <input type="number" name="n" required>
        <input type="submit" value="Generate">
    </form>
    '''

@app.route('/even')
def even():
    n = int(request.args.get('n'))

    numbers = [2 * i for i in range(1, n + 1)]

    return f"<h2>First {n} Even Numbers:</h2>{numbers}"

if __name__ == '__main__':
    app.run(host='127.0.0.1', port=8080)

from flask import Flask, request

app = Flask(__name__)

@app.route('/')
def home():
    return '''
    <h2>Matrix Multiplication</h2>
    <form action="/multiply" method="get">
        Enter matrix elements (2x2):
        <p>Matrix A (comma separated):</p>
        <input name="a" placeholder="1,2,3,4" required>

        <p>Matrix B (comma separated):</p>
        <input name="b" placeholder="5,6,7,8" required>

        <br><br>
        <input type="submit" value="Multiply">
    </form>
    '''

@app.route('/multiply')
def multiply():
    a = list(map(int, request.args.get('a').split(',')))
    b = list(map(int, request.args.get('b').split(',')))

    if len(a) != 4 or len(b) != 4:
        return "Enter exactly 4 elements for each matrix."

    A = [a[:2], a[2:]]
    B = [b[:2], b[2:]]

    C = [[0, 0], [0, 0]]

    for i in range(2):
        for j in range(2):
            for k in range(2):
                C[i][j] += A[i][k] * B[k][j]

    return f"<h2>Result of Matrix Multiplication:</h2>{C}"

if __name__ == '__main__':
    app.run(host='127.0.0.1', port=8080)
